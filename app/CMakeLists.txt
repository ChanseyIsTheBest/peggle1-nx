# Peggle (Xbox Live Arcade, title 58410889, with Title Update #3): static recompilation host application.
#
# The game code is not in this repository: run the code generator first (tools/codegen.ps1), which writes generated/
# from your own default.xex together with the CMake glue (generated/rexglue.cmake, made by `rexglue codegen`).
#
# Windows PC:
#   cmake --preset win-pc && cmake --build --preset win-pc        (from app/, inside tools\win\env.ps1)
# Nintendo Switch (devkitA64), later:
#   cmake -S app -B out/nx -G Ninja -DCMAKE_BUILD_TYPE=Release \
#     -DCMAKE_TOOLCHAIN_FILE=tools/switch/cmake/switch-devkitA64.cmake -DREXGLUE_SWITCH_NVK_SDK=<mesa sdk>

# Console, 2026-10-13: GCC's AArch64 early register allocation pass (early_ra; -mearly-ra=all is the default from
# -O2 in devkitA64's GCC 16.1) miscompiled the game's streaming copies for an unaligned source (sub_8260E9C0,
# sub_8260E164: 72 two-register table lookups, 65 vectors live). The optimised GIMPLE was right; after the pass 38
# of the lookups were gone, the register with the byte-reversal constant was overwritten by a data load inside the
# loop, and every atlas copied through them was garbage from its second megabyte on. With the pass off the function
# is complete at -O2 and -O3 (docs/peggle1-notes.md, "Console: large atlases"). Off for everything built for the
# console, the SDK included (directory scope, before the SDK is added), and at the link, where an LTO build
# generates its code. tools/check_switch_build.py checks that every compile command has it.
if(REXGLUE_PLATFORM_SWITCH AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND NOT CMAKE_CXX_COMPILER_VERSION VERSION_LESS 14)
    add_compile_options(-mearly-ra=none)
    add_link_options(-mearly-ra=none)
endif()

# The SDK compiles itself with these (directory scope), but they do not reach the generated game code, which is
# built in this directory. GCC's default for GNU C++ is -ffp-contract=fast: it fuses separate PPC fmul/fadd into
# FMA and results drift from the Xbox 360. Strict aliasing is unsafe for recompiled code (guest memory is read and
# written at every width through the same addresses). Checked in build.ninja 2026-10-06: the Switch objects had both;
# the PC (clang) ones had neither, so Clang is included now (banjo-nx's check, relayed by the Mushihimesama session).
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    foreach(target peggle1 peggle1_recomp)
        if(TARGET ${target})
            target_compile_options(${target} PRIVATE -ffp-contract=off -fno-strict-aliasing)
        endif()
    endforeach()
    # 2026-10-07 (masseffect-nx ships them; futari-nx: menus 426/427 and its fade 8/8 identical, verify 0): sqrt and
    # friends without the errno branch, FP operations that may be if-converted or vectorised. The guest's FPSCR here
    # is only the rounding and flush modes (sdk/include/rex/ppc/context.h): no FP exception flag is ever read.
    # OFF in Peggle 2 until a quiet-PC lockstep settles it: two captures with them missed the menu fade-out of game
    # frames 1022-1047 that the CPU-off baselines reproduce, but a capture WITHOUT them made under the same load (a
    # Switch LTO build running) missed it too, so the cause is not shown to be the flags. Verify mode: 0 differences.
    option(PEGGLE1_NO_MATH_ERRNO "-fno-math-errno -fno-trapping-math on the game code and the app" OFF)
    if(PEGGLE1_NO_MATH_ERRNO)
        foreach(target peggle1 peggle1_recomp)
            if(TARGET ${target})
                target_compile_options(${target} PRIVATE -fno-math-errno -fno-trapping-math)
            endif()
        endforeach()
    endif()
    # sa2-nx's second CPU round (2026-10-12), the game code only: `sqrt` without the errno check (a compare and a
    # branch to the library after every fsqrt). Values are unchanged: nothing reads errno. The same flag on the file
    # with the localized copies (GCC does not inline between two float functions that differ in -fmath-errno).
    # -fno-trapping-math stays out (sa2-nx's audit: it lets the compiler run a float operation earlier than the code
    # does, across a flush-mode switch it does not know the operation depends on).
    option(PEGGLE1_GAME_NO_MATH_ERRNO "-fno-math-errno on the recompiled game code and the localized copies" ON)
    if(PEGGLE1_GAME_NO_MATH_ERRNO AND NOT PEGGLE1_NO_MATH_ERRNO)
        if(TARGET peggle1_recomp)
            target_compile_options(peggle1_recomp PRIVATE -fno-math-errno)
        endif()
        set_property(SOURCE src/cpu/peggle1_hot_guest.cpp APPEND PROPERTY COMPILE_OPTIONS -fno-math-errno)
    endif()
endif()

# Switch compiler set (sa2-nx's round 3, UnleashedRecomp-NX's set without PGO; the user's choice for the ports):
#   - LTO: the recompiled game (peggle1_recomp) and the app (renderer, natives) are compiled to GIMPLE
#     (-flto -fno-fat-lto-objects) and optimised as one program at the link (-flto=N -flto-partition=balanced, N
#     parallel LTRANS jobs, kept low for the build machine). Calls between unhooked game functions are direct
#     (tools/perf_codegen_post.py), so small game functions inline across the 122 generated files. SDK libraries stay out.
#   - -fipa-pta at compile and link (interprocedural points-to analysis).
#   - The link repeats the exactness flags (the LTRANS stage must not fuse or alias what the compile stage kept apart),
#     and xxHash is built with XXH_FORCE_MEMORY_ACCESS=0 (nfsmw-nx's LTO pitfall: its default reads violate strict
#     aliasing once inlined into a key builder).
#   - Line tables (-g1, DWARF 4) for the app's own code: console profiles map to source lines on the archived ELF; -g
#     does not change GCC's code generation.
if(REXGLUE_PLATFORM_SWITCH)
    # peggle1-nx (2026-10-21): off, releases included (the user: "We don't need LTO or PGO because the game is extremely
    # lightweight").
    option(PEGGLE1_SWITCH_LTO "Link-time optimisation of the game code and the app (Switch)" OFF)
    set(PEGGLE1_SWITCH_LTO_JOBS "2" CACHE STRING "Parallel LTRANS jobs of the Switch LTO link")
    option(PEGGLE1_SWITCH_IPA_PTA "-fipa-pta at the Switch LTO link" ON)
    option(PEGGLE1_SWITCH_LINE_INFO "Line tables (-g1) for the app's own code in the Switch ELF" ON)
    target_compile_definitions(peggle1 PRIVATE XXH_FORCE_MEMORY_ACCESS=0)
    if(PEGGLE1_SWITCH_LTO)
        set(_peggle1_lto_compile -flto -fno-fat-lto-objects)
        set(_peggle1_lto_link -flto=${PEGGLE1_SWITCH_LTO_JOBS} -flto-partition=balanced -ffp-contract=off
            -fno-strict-aliasing)
        if(PEGGLE1_NO_MATH_ERRNO)
            list(APPEND _peggle1_lto_link -fno-math-errno -fno-trapping-math)
        endif()
        if(PEGGLE1_SWITCH_IPA_PTA)
            list(APPEND _peggle1_lto_compile -fipa-pta)
            list(APPEND _peggle1_lto_link -fipa-pta)
        endif()
        foreach(target peggle1_recomp peggle1)
            if(TARGET ${target})
                target_compile_options(${target} PRIVATE ${_peggle1_lto_compile})
            endif()
        endforeach()
        target_link_options(peggle1 PRIVATE ${_peggle1_lto_link})
        message(STATUS "peggle1: Switch LTO on (${PEGGLE1_SWITCH_LTO_JOBS} LTRANS jobs, -fipa-pta ${PEGGLE1_SWITCH_IPA_PTA})")
    endif()
    if(PEGGLE1_SWITCH_LINE_INFO)
        target_compile_options(peggle1 PRIVATE -g1 -gdwarf-4)
        target_link_options(peggle1 PRIVATE -g1 -gdwarf-4)
    endif()
    # sasrt-nx's CPU round through sa2-nx (2026-10-12). Each has its switch:
    #   - No frame pointer in the game code: AArch64 GCC keeps the x29 frame chain in every non-leaf function by
    #     default, which costs the set-up instruction and one of the callee-saved registers the recompiled code is
    #     short of. The SDK and the app keep theirs (the console profiler's caller chains end at a game function).
    #   - The flush-mode checks the control flow fixes compile to nothing here (tools/flush_mode_flow.py); OFF defines
    #     REX_FLUSH_FLOW_OFF, which puts the original checks back.
    #   - Guest compare-and-swap without the legacy builtin's trailing `dmb ish` (tools/guest_cas.py); OFF defines
    #     REX_GUEST_CAS_LEGACY (the run-time key peggle1_cpu_guest_cas_no_barrier puts the barrier back without a build).
    #   - Guest vector loads and stores reversed with REV64 + EXT instead of a table lookup (tools/vec_byteswap.py); OFF
    #     defines REX_BSWAP128_TABLE.
    #   - Constant word shuffles as one permutation (tools/pch_neon_shuffles.py); OFF defines REX_SIMDE_SHUFFLES.
    #   - vperm and the float-to-integer conversions as their NEON instruction (rex/ppc/intrinsics.h); OFF defines
    #     REX_VMX_HELPERS_X86_FORM.
    option(PEGGLE1_SWITCH_OMIT_FRAME_POINTER "Game code without the x29 frame chain (Switch)" ON)
    option(PEGGLE1_SWITCH_FLUSH_FLOW "Flush-mode checks proven by control flow compile to nothing (Switch)" ON)
    option(PEGGLE1_SWITCH_GUEST_CAS "Guest compare-and-swap without the trailing dmb (Switch)" ON)
    option(PEGGLE1_SWITCH_VEC_BYTESWAP "Guest vector byte reversal as REV64 + EXT (Switch)" ON)
    option(PEGGLE1_SWITCH_NEON_SHUFFLES "Constant word shuffles as one permutation (Switch)" ON)
    option(PEGGLE1_SWITCH_NEON_VMX_HELPERS "vperm and float-to-integer conversions as NEON instructions (Switch)" ON)
    set(_peggle1_round1_definitions "")
    if(NOT PEGGLE1_SWITCH_FLUSH_FLOW)
        list(APPEND _peggle1_round1_definitions REX_FLUSH_FLOW_OFF)
    endif()
    if(NOT PEGGLE1_SWITCH_GUEST_CAS)
        list(APPEND _peggle1_round1_definitions REX_GUEST_CAS_LEGACY)
    endif()
    if(NOT PEGGLE1_SWITCH_VEC_BYTESWAP)
        list(APPEND _peggle1_round1_definitions REX_BSWAP128_TABLE)
    endif()
    if(NOT PEGGLE1_SWITCH_NEON_SHUFFLES)
        list(APPEND _peggle1_round1_definitions REX_SIMDE_SHUFFLES)
    endif()
    if(NOT PEGGLE1_SWITCH_NEON_VMX_HELPERS)
        list(APPEND _peggle1_round1_definitions REX_VMX_HELPERS_X86_FORM)
    endif()
    foreach(target peggle1_recomp peggle1)
        if(TARGET ${target} AND _peggle1_round1_definitions)
            target_compile_definitions(${target} PRIVATE ${_peggle1_round1_definitions})
        endif()
    endforeach()
    if(PEGGLE1_SWITCH_OMIT_FRAME_POINTER)
        if(TARGET peggle1_recomp)
            target_compile_options(peggle1_recomp PRIVATE -fomit-frame-pointer)
        endif()
        # The localized copies (app/generated/hot, all compiled in one file of the app) are game code too.
        set_property(SOURCE src/cpu/peggle1_hot_guest.cpp APPEND PROPERTY COMPILE_OPTIONS -fomit-frame-pointer)
    endif()
    message(STATUS "peggle1: CPU round 1: omit frame pointer ${PEGGLE1_SWITCH_OMIT_FRAME_POINTER}, flush flow "
                   "${PEGGLE1_SWITCH_FLUSH_FLOW}, guest CAS ${PEGGLE1_SWITCH_GUEST_CAS}, vector byteswap "
                   "${PEGGLE1_SWITCH_VEC_BYTESWAP}, NEON shuffles ${PEGGLE1_SWITCH_NEON_SHUFFLES}, NEON VMX helpers "
                   "${PEGGLE1_SWITCH_NEON_VMX_HELPERS}")
endif()

if(REXGLUE_PLATFORM_SWITCH)
    # Profile-guided optimisation (2026-10-14, the user's request for the 1.0.0 release; sa2-nx's setup), for the game
    # code and the app like LTO:
    #   PEGGLE1_PGO=generate  an instrumented NRO that writes its branch counts to pgo/ on the SD card
    #                         (src/peggle1_pgo.cpp; registered in a section, not by constructors: libgcov's exit-time
    #                         files do not fit the Switch). Value profiling is off: its shared lists are not safe
    #                         without atomic updates.
    #   PEGGLE1_PGO=use       the counts merged on the PC (tools/win/pgo_merge.ps1). Code the sessions did not run is
    #                         compiled as without a profile (-fprofile-partial-training), and functions changed since
    #                         the instrumented build keep no profile (-Wno-coverage-mismatch).
    # The profile is pgo/ at the root of the repository (PEGGLE1_PGO_DIR): one .gcda per object, named by the object's
    # path in the build folder with # for each folder separator. -fprofile-prefix-path makes GCC use those relative
    # names, so the profile works from any checkout; it must be the build folder written the Windows way, as GCC
    # sees it.
    set(PEGGLE1_PGO "" CACHE STRING "Profile-guided optimisation of the Switch build: empty, generate or use")
    set_property(CACHE PEGGLE1_PGO PROPERTY STRINGS "" generate use)
    set(PEGGLE1_PGO_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../pgo" CACHE PATH "Folder of the PGO profile (.gcda files)")
    get_filename_component(_peggle1_pgo_dir "${PEGGLE1_PGO_DIR}" ABSOLUTE)
    file(TO_NATIVE_PATH "${_peggle1_pgo_dir}" _peggle1_pgo_dir)
    file(TO_NATIVE_PATH "${CMAKE_BINARY_DIR}" _peggle1_pgo_prefix)
    if(PEGGLE1_PGO STREQUAL "generate")
        set(_peggle1_pgo "-fprofile-generate=${_peggle1_pgo_dir}" "-fprofile-prefix-path=${_peggle1_pgo_prefix}"
                         -fno-profile-values -fprofile-update=single -fprofile-info-section=gcov_info)
        target_compile_definitions(peggle1 PRIVATE PEGGLE1_PGO_GENERATE=1)
    elseif(PEGGLE1_PGO STREQUAL "use")
        if(NOT EXISTS "${PEGGLE1_PGO_DIR}")
            message(FATAL_ERROR "PEGGLE1_PGO=use: no profile in ${PEGGLE1_PGO_DIR}")
        endif()
        # -fprofile-correction: the counts were taken without atomic updates (-fprofile-update=single) while several
        # game threads ran the same code, so a few increments were lost and some blocks' counts do not add up; GCC
        # smooths them instead of stopping.
        set(_peggle1_pgo "-fprofile-use=${_peggle1_pgo_dir}" "-fprofile-prefix-path=${_peggle1_pgo_prefix}"
                         -fprofile-correction -fno-profile-values -fprofile-partial-training -Wno-missing-profile
                         -Wno-coverage-mismatch)
        # sa2-nx's inliner limits for a build with a profile (measured there: 41 % fewer executed calls for the hot
        # functions 17 % larger; without the unit growth limit the higher one fills the default 40 % with code the
        # session never ran). Empty = the compiler's.
        set(PEGGLE1_SWITCH_INLINE_AUTO "64" CACHE STRING
            "max-inline-insns-auto of the game code with a profile (empty: the compiler's)")
        set(PEGGLE1_SWITCH_INLINE_UNIT_GROWTH "12" CACHE STRING
            "inline-unit-growth of the game code with a profile (empty: the compiler's)")
        set(_peggle1_inline_options "")
        if(PEGGLE1_SWITCH_INLINE_AUTO)
            list(APPEND _peggle1_inline_options --param=max-inline-insns-auto=${PEGGLE1_SWITCH_INLINE_AUTO})
        endif()
        if(PEGGLE1_SWITCH_INLINE_UNIT_GROWTH)
            list(APPEND _peggle1_inline_options --param=inline-unit-growth=${PEGGLE1_SWITCH_INLINE_UNIT_GROWTH})
        endif()
        if(_peggle1_inline_options AND TARGET peggle1_recomp)
            target_compile_options(peggle1_recomp PRIVATE ${_peggle1_inline_options})
            set_property(SOURCE src/cpu/peggle1_hot_guest.cpp APPEND PROPERTY COMPILE_OPTIONS ${_peggle1_inline_options})
        endif()
    elseif(PEGGLE1_PGO)
        message(FATAL_ERROR "PEGGLE1_PGO must be empty, generate or use (got ${PEGGLE1_PGO})")
    endif()
    if(PEGGLE1_PGO)
        foreach(target peggle1_recomp peggle1)
            if(TARGET ${target})
                target_compile_options(${target} PRIVATE ${_peggle1_pgo})
            endif()
        endforeach()
        target_link_options(peggle1 PRIVATE ${_peggle1_pgo})
        message(STATUS "peggle1: PGO ${PEGGLE1_PGO}")
    endif()

    # The Switch build's fixed settings, compiled in (src/peggle1_app.h BuiltInConfig): everything peggle1.toml no
    # longer lists. The SD card's peggle1.toml holds the player's options and overrides these key by key. Both files
    # are written by tools/make_enhancement_configs.py. (sa2-nx's setup.)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/peggle1_switch_defaults.toml" PEGGLE1_BUILTIN_CONFIG)
    # And the player's peggle1.toml, written next to the NRO when it is missing (src/peggle1_app.h DefaultConfigFile).
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/peggle1_switch.toml" PEGGLE1_PLAYER_CONFIG)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
                 "${CMAKE_CURRENT_SOURCE_DIR}/peggle1_switch_defaults.toml"
                 "${CMAKE_CURRENT_SOURCE_DIR}/peggle1_switch.toml")
    configure_file("${CMAKE_CURRENT_SOURCE_DIR}/src/peggle1_builtin_config.h.in"
                   "${CMAKE_CURRENT_BINARY_DIR}/builtin_config/peggle1_builtin_config.h" @ONLY)
    target_include_directories(peggle1 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/builtin_config")
    target_compile_definitions(peggle1 PRIVATE PEGGLE1_BUILTIN_CONFIG=1)

    # Link map (from futari-nx): which function and object each address of the NRO belongs to, for
    # tools/switch_hot_functions.py and the console profiler's samples. It must come from the same build as the NRO.
    target_link_options(peggle1 PRIVATE "LINKER:-Map=${CMAKE_BINARY_DIR}/peggle1.map")
    # The homebrew menu's entry: name, author, version and icon (app/icon.jpg, 256x256 JPEG).
    set(PEGGLE1_VERSION "1.0.0" CACHE STRING "NRO display version" FORCE)
    # peggle1-nx (2026-10-21): app/icon.jpg is the game's own 64x64 icon.png scaled up, the one piece of the game's art
    # in the repository (the user's exception for the icon). out/icon_dev.jpg is the same picture for older trees.
    set(_peggle1_icon "")
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/icon.jpg")
        set(_peggle1_icon ICON "${CMAKE_CURRENT_SOURCE_DIR}/icon.jpg")
    elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../out/icon_dev.jpg")
        set(_peggle1_icon ICON "${CMAKE_CURRENT_SOURCE_DIR}/../out/icon_dev.jpg")
    endif()
    rexglue_switch_add_nro(peggle1
        NAME "Peggle"
        AUTHOR "ChanseyIsTheBest"
        VERSION "${PEGGLE1_VERSION}"
        ${_peggle1_icon})
endif()
